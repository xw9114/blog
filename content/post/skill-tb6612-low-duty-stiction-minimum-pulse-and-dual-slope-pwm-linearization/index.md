{

        return 0.0f;
    }

    return kick + ((u - u_dead) / (1.0f - u_dead)) * (dmax - kick);
}

void Tb6612_Update(Tb6612Motor_t *motor,
                   float command,
                   float vbat_v,
                   float measured_speed_abs,
                   uint32_t now_us,
                   bool brake_when_zero,
                   Tb6612DriveDiag_t *diag)
{
    const float cmd = Tb6612_ClampF(command, -TB6612_MAX_COMMAND_ABS, TB6612_MAX_COMMAND_ABS);
    const float cmd_abs = fabsf(cmd);
    const int8_t requested_dir = (cmd > TB6612_ZERO_EPS) ? 1 : ((cmd < -TB6612_ZERO_EPS) ? -1 : 0);
    float duty_nominal = 0.0f;
    float duty_comp = 0.0f;
    uint32_t ccr = 0U;

    motor->in_deadband = (cmd_abs <= motor->dead_command) || (requested_dir == 0);
    motor->reversal_guard_active = false;

    if (motor->in_deadband)
    {
        motor->kick_active = false;
        motor->kick_until_us = 0U;
        motor->pending_dir = 0;
        motor->applied_dir = 0;

        Tb6612_SetDirectionPins(motor, 0, brake_when_zero);
        ccr = Tb6612_WritePwm(motor, 0.0f);
        goto publish_diag;
    }

    /* ????????? PWM???????????????? H ?????????? */
    if ((motor->applied_dir != 0) && (requested_dir != motor->applied_dir))
    {
        motor->pending_dir = requested_dir;
        motor->applied_dir = 0;
        motor->last_dir_change_us = now_us;
        motor->kick_active = false;

        Tb6612_SetDirectionPins(motor, 0, true);
        ccr = Tb6612_WritePwm(motor, 0.0f);
        motor->reversal_guard_active = true;
        goto publish_diag;
    }

    if ((motor->pending_dir != 0) && (motor->applied_dir == 0))
    {
        if (Tb6612_ElapsedUs(now_us, motor->last_dir_change_us) < TB6612_DIR_REVERSAL_GUARD_US)
        {
            Tb6612_SetDirectionPins(motor, 0, true);
            ccr = Tb6612_WritePwm(motor, 0.0f);
            motor->reversal_guard_active = true;
            goto publish_diag;
        }

        motor->applied_dir = motor->pending_dir;
        motor->pending_dir = 0;
    }

    if (motor->applied_dir == 0)
    {
        motor->applied_dir = requested_dir;
    }

    Tb6612_SetDirectionPins(motor, motor->applied_dir, false);

    duty_nominal = Tb6612_MapCommandToDuty(motor, cmd_abs);

    /* ???????????????????????????????? */

        motor->kick_active = true;
        motor->kick_until_us = now_us + motor->kick_time_us;
    }

    if (motor->kick_active)
    {
        if ((int32_t)(now_us - motor->kick_until_us) < 0)
        {
            duty_nominal = fmaxf(duty_nominal, motor->kick_duty);
        }
        else
        {
            motor->kick_active = false;
        }
    }

    /* ???????????????????????????? */
    if ((measured_speed_abs > 20.0f) && (requested_dir != motor->applied_dir))
    {
        duty_nominal = fminf(duty_nominal, motor->brake_duty_limit);
    }

    duty_comp = Tb6612_CompensateBattery(duty_nominal, vbat_v, motor->max_duty);
    ccr = Tb6612_WritePwm(motor, duty_comp);

publish_diag:
    if (diag != NULL)
    {
        diag->requested_command = cmd;
        diag->mapped_duty = duty_nominal;
        diag->battery_compensated_duty = duty_comp;
        diag->compare_counts = ccr;
        diag->in_deadband = motor->in_deadband;
        diag->kick_active = motor->kick_active;
        diag->reversal_guard_active = motor->reversal_guard_active;
        diag->direction = motor->applied_dir;
    }
}
